  LAST_ORDER_MINUTE = 0

  LAST_ORDER_HOUR = 14
  LAST_ORDER_MINUTE = 30

To change market close to 4:00 PM:
  MARKET_CLOSE_HOUR = 16
  MARKET_CLOSE_MINUTE = 0
